# Vue3 排序算法可视化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用 Vite + Vue3 + TS 做单页排序可视化：可填数量生成柱子、五种排序逐步动画、统计移动次数与含动画耗时，支持停止/打乱。

**Architecture:** 五种算法在 `sorts.ts` 对高度数组副本预生成 `swap`/`write` 步骤队列；`App.vue` 负责生成柱子、逐步播放、高亮、abort 与统计。不拆多余组件。

**Tech Stack:** Vite、Vue 3、TypeScript、原生 CSS（无 UI 库）

**Spec:** `docs/superpowers/specs/2026-10-26-Vue3排序算法可视化.md`

## Global Constraints

- 算法仅限：冒泡、选择、插入、快速、归并
- 移动次数 = `swap` + `write` 步数；不做比较次数
- 每步约 `40ms` delay + 高度 `transition 0.03s`
- 数量合法范围 `2～500`，默认输入 `100`
- UI 色号/布局以 spec §3.1 为准（柱 `#3B82F6`，高亮 `#F59E0B`）
- 首版不引入单元测试框架；`sorts` 用临时 node 脚本校验，UI 按 spec §8 手动点验
- 人类可读：主路径只在 `App.vue` + `sorts.ts`，不抽无必要 composable

```ts
export type Step =
  | { type: 'swap'; i: number; j: number }
  | { type: 'write'; i: number; value: number }

export function applySteps(arr: number[], steps: Step[]): number[] {
  const a = [...arr]
  for (const step of steps) {
    if (step.type === 'swap') {
      const t = a[step.i]
      a[step.i] = a[step.j]
      a[step.j] = t
    } else {
      a[step.i] = step.value
    }
  }
  return a
}

export function bubbleSortSteps(arr: number[]): Step[] {
  const a = [...arr]
  const steps: Step[] = []
  for (let n = a.length; n > 1; n--) {
    for (let i = 0; i < n - 1; i++) {
      if (a[i] > a[i + 1]) {
        const t = a[i]
        a[i] = a[i + 1]
        a[i + 1] = t
        steps.push({ type: 'swap', i, j: i + 1 })
      }
    }
  }
  return steps
}

export function selectionSortSteps(arr: number[]): Step[] {
  const a = [...arr]
  const steps: Step[] = []
  for (let i = 0; i < a.length; i++) {
    let min = i
    for (let j = i + 1; j < a.length; j++) {
      if (a[j] < a[min]) min = j
    }
    if (min !== i) {
      const t = a[i]
      a[i] = a[min]
      a[min] = t
      steps.push({ type: 'swap', i, j: min })
    }
  }
  return steps
}

export function insertionSortSteps(arr: number[]): Step[] {
  const a = [...arr]
  const steps: Step[] = []
  for (let i = 1; i < a.length; i++) {
    const key = a[i]
    let j = i - 1
    while (j >= 0 && a[j] > key) {
      a[j + 1] = a[j]
      steps.push({ type: 'write', i: j + 1, value: a[j] })
      j--
    }
    a[j + 1] = key
    steps.push({ type: 'write', i: j + 1, value: key })
  }
  return steps
}

export function quickSortSteps(arr: number[]): Step[] {
  const a = [...arr]
  const steps: Step[] = []

  function swap(i: number, j: number) {
    if (i === j) return
    const t = a[i]
    a[i] = a[j]
    a[j] = t
    steps.push({ type: 'swap', i, j })
  }

  function partition(lo: number, hi: number): number {
    const pivot = a[hi]
    let i = lo
    for (let j = lo; j < hi; j++) {
      if (a[j] < pivot) {
        swap(i, j)
        i++
      }
    }
    swap(i, hi)
    return i
  }

  function sort(lo: number, hi: number) {
    if (lo >= hi) return
    const p = partition(lo, hi)
    sort(lo, p - 1)
    sort(p + 1, hi)
  }

  sort(0, a.length - 1)
  return steps
}

export function mergeSortSteps(arr: number[]): Step[] {
  const a = [...arr]
  const steps: Step[] = []

  function merge(lo: number, mid: number, hi: number) {
    const left = a.slice(lo, mid + 1)
    const right = a.slice(mid + 1, hi + 1)
    let i = 0
    let j = 0
    let k = lo
    while (i < left.length && j < right.length) {
      if (left[i] <= right[j]) {
        a[k] = left[i]
        steps.push({ type: 'write', i: k, value: left[i] })
        i++
      } else {
        a[k] = right[j]
        steps.push({ type: 'write', i: k, value: right[j] })
        j++
      }
      k++
    }
    while (i < left.length) {
      a[k] = left[i]
      steps.push({ type: 'write', i: k, value: left[i] })
      i++
      k++
    }
    while (j < right.length) {
      a[k] = right[j]
      steps.push({ type: 'write', i: k, value: right[j] })
      j++
      k++
    }
  }

  function sort(lo: number, hi: number) {
    if (lo >= hi) return
    const mid = Math.floor((lo + hi) / 2)
    sort(lo, mid)
    sort(mid + 1, hi)
    merge(lo, mid, hi)
  }

  sort(0, a.length - 1)
  return steps
}
```

```ts
  for (let i = list.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1))
    const t = list[i]
    list[i] = list[j]
    list[j] = t
  }
}

async function generateBars() {
  const n = Number(countInput.value)
  if (!Number.isInteger(n) || n < 2 || n > 500) {
    countError.value = '请输入 2～500 的整数'
    return
  }
  countError.value = ''

  if (running.value) await stopCurrentPlay()

  await nextTick()
  const el = chartEl.value
  if (!el) return
  const maxH = Math.max(MIN_BAR_H + 1, el.clientHeight - 12)
  const step = (maxH - MIN_BAR_H) / Math.max(1, n - 1)

  const next: Bar[] = []
  for (let i = 0; i < n; i++) {
    next.push({ id: i, height: Math.round(MIN_BAR_H + step * i) })
  }
  shuffleInPlace(next)
  // 打乱后重赋稳定 id（按下标），交换时只改 height、key 固定在槽位
  bars.value = next.map((b, index) => ({ id: index, height: b.height }))
  hasBars.value = true
  resetStats()
}

async function shuffleBars() {
  if (!hasBars.value) return
  if (running.value) await stopCurrentPlay()
  const heights = bars.value.map((b) => b.height)
  shuffleInPlace(heights)
  bars.value = bars.value.map((b, i) => ({ id: b.id, height: heights[i] }))
  resetStats()
}
```

```ts
import {
  bubbleSortSteps,
  selectionSortSteps,
  insertionSortSteps,
  quickSortSteps,
  mergeSortSteps,
  type Step,
} from './sorts'

const sortFns = {
  bubble: bubbleSortSteps,
  selection: selectionSortSteps,
  insertion: insertionSortSteps,
  quick: quickSortSteps,
  merge: mergeSortSteps,
} as const

type SortName = keyof typeof sortFns

function delay(ms: number) {
  return new Promise<void>((resolve) => setTimeout(resolve, ms))
}

function applyStepToBars(step: Step) {
  const list = bars.value
  if (step.type === 'swap') {
    const h = list[step.i].height
    list[step.i] = { ...list[step.i], height: list[step.j].height }
    list[step.j] = { ...list[step.j], height: h }
    activeIndexes.value = [step.i, step.j]
  } else {
    list[step.i] = { ...list[step.i], height: step.value }
    activeIndexes.value = [step.i]
  }
}

async function runSort(name: SortName) {
  if (!hasBars.value || running.value) return

  abort = false
  const token = ++playToken
  running.value = true
  status.value = '进行中'
  moves.value = 0
  elapsedText.value = '—'

  const heights = bars.value.map((b) => b.height)
  const steps = sortFns[name](heights)
  const t0 = performance.now()

  for (const step of steps) {
    if (abort || token !== playToken) break
    applyStepToBars(step)
    moves.value += 1
    await delay(STEP_MS)
  }

  const elapsed = (performance.now() - t0) / 1000
  elapsedText.value = `${elapsed.toFixed(2)} s`
  activeIndexes.value = []
  running.value = false

  if (abort || token !== playToken) {
    status.value = '已停止'
  } else {
    status.value = '已完成'
  }
}

function stopSort() {
  if (!running.value) return
  abort = true
}
```
